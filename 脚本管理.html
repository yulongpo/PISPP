function iq_data = generate_qpsk(params)

% 输入:

%     params - 结构体，包含以下字段:

%         .sample_rate  : 采样率 (Hz), 默认 1e6

%         .symbol_rate  : 符号速率 (Hz), 默认 100e3

%         .carrier_freq : 载波频率 (Hz), 默认 2.4e6

%         .mod_type     : 调制类型 ('qpsk', 'bpsk', '16qam'), 默认 'qpsk'


         % 设置默认参数

         if nargin < 1, params = struct(); end

         snr_db        = getfield(params, 'snr_db', 10);

         n_samples     = getfield(params, 'samples', 1024);

         fs            = getfield(params, 'sample_rate', 1e6);

         rs            = getfield(params, 'symbol_rate', 100e3);

         fc            = getfield(params, 'carrier_freq', 2.4e6);

         mod_type      = lower(getfield(params, 'mod_type', 'qpsk'));


         % 生成随机符号

        switch mod_type

              case 'bpsk'

                   M =2;

                   symbols = randi([0 M-1], ceil(n_samples * rs / fs), 1);

                   mapping = exp(1j * 2*pi * (0:M-1)/M);  % BPSK: 0°, 180°

              case 'qpsk'

                   M = 4;

                   symbols = randi([0 M-1], ceil(n_samples * rs / fs), 1);
